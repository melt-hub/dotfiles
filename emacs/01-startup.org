** Paths and Package Manager Setup

  Core directory paths and registration of the `use-package` engine path.

  #+begin_src emacs-lisp
    ;; Core System Paths
    (defvar os-packages-path "~/dotfiles/emacs/packages/"
      "Directory containing local/vendor package subdirectories.")

    (defvar org-dir-path "~/zk/"
      "Root directory for Org-mode notes and Zettelkasten.")

    (defvar my/temp-dir (expand-file-name ".tmp/" user-emacs-directory)
      "Central directory for Emacs temporary files.")

    ;; Load use-package path prior to loading modules
    (add-to-list 'load-path (concat os-packages-path "use-package/"))
  #+end_src

      "Default university for scientific papers.")

** System Defaults and Environment

  Global options for encodings, UI, compilation behavior, and security prompts.

  #+begin_src emacs-lisp
    ;; Encoding and Buffer Defaults
    (prefer-coding-system 'utf-8-unix)
    (setq inhibit-startup-screen t)
    (setq vc-follow-symlinks t)
    (setq epa-pinentry-mode 'loopback)

    ;; Global Compilation Behavior
    (setq compilation-scroll-output t)
    (add-hook 'compilation-filter-hook 'ansi-color-compilation-filter)

    (my/say "System variables and paths initialized.")
  #+end_src

** File Mode Associations

  Automatic major mode mappings for generic configuration and system files.

  #+begin_src emacs-lisp
    ;; Generic and System File Associations
    (add-to-list 'auto-mode-alist '("\\.ino\\'" . c++-mode))
    (add-to-list 'auto-mode-alist '("\\.rasi\\'" . css-mode))
    (add-to-list 'auto-mode-alist '("/config\\'" . conf-mode))
    (add-to-list 'auto-mode-alist '("\\.*rc\\'" . conf-mode))

    (my/say "System file automodes applied.")
  #+end_src
