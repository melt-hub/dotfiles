#+TITLE: Emacs Startup Configuration
#+PROPERTY: header-args :tangle "~/.emacs.d/modules/my-startup.el"

* Startup

  This section contains performance-oriented settings, path registrations, and
  core variables initialized during the Emacs startup phase.

  #+begin_src emacs-lisp
    ;; ====================| STARTUP |====================

    ;; core system paths
    (defvar org-dir-path "~/zk/"
      "Root directory for Org-mode notes and Zettelkasten.")

    (defvar my/temp-dir (expand-file-name ".tmp/" user-emacs-directory)
      "Central directory for Emacs temporary files.")

    ;; system defaults and environment
    (setq inhibit-startup-screen t)
    (setq vc-follow-symlinks t)
    (setq epa-pinentry-mode 'loopback)

    ;; global compilation behavior
    (setq compilation-scroll-output t)
    (add-hook 'compilation-filter-hook #'ansi-color-compilation-filter)

    ;; file mode associations
    (add-to-list 'auto-mode-alist '("\\.ino\\'" . c++-mode))
    (add-to-list 'auto-mode-alist '("\\.rasi\\'" . css-mode))
    (add-to-list 'auto-mode-alist '("/config\\'" . conf-mode))
    (add-to-list 'auto-mode-alist '("\\..*rc\\'" . conf-mode))

    ;; module finalization
    (provide 'my-startup)
    ;; ====================| END STARTUP |====================
  #+end_src
