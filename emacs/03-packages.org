** Initialization

  Configurations for package managers, archive priorities, and load paths.

  #+begin_src emacs-lisp
    ;; package archives configuration
    (use-package package
      :config
      (setq package-archive-priorities
            '(("melpa-stable" . 2)
              ("MELPA" . 1)
              ("gnu" . 0)))
      (setq package-archives
            '(("melpa-stable" . "https://stable.melpa.org/packages/")
              ("MELPA" . "https://melpa.org/packages/")
              ("gnu" . "https://elpa.gnu.org/packages/"))))

    (setq use-package-compute-statistics t)
    (require 'use-package)
  #+end_src

** Org

  Packages focused on text editing, task management, knowledge graphs, and

*** Org Mode

  A versatile note-taking, planning, and document-exporting system.

  #+begin_src emacs-lisp
    ;; org mode core configuration
    (use-package org
      :defer t
      :bind (("C-c a" . org-agenda) ("C-c c" . org-capture))
      :config
      (setq org-startup-with-inline-images t)

      (add-to-list 'org-structure-template-alist '("L" . "latex"))
      (add-to-list 'org-src-lang-modes '("latex" . LaTeX))

      (let ((agenda-path "/home/melt/zk/agenda/agenda.org"))
        (setq org-agenda-files (list agenda-path))
        (setq org-log-done 'time)
        (setq org-log-into-drawer t)

        (setq org-tag-alist
              '((:startgroup)
                ("task" . ?k) ("rem"  . ?r) ("bday" . ?b)
                (:endgroup)))

        (setq org-capture-templates
              `(("k" "task" entry
                 (file+headline ,agenda-path "Tasks & Appointments")
                 ,(concat "* TODO %^{Description} :task:\nSCHEDULED: %^t\n"
                          ":PROPERTIES:\n:CONTACT: %^{Who}\n"
                          ":LOCATION: %^{Where}\n:CREATED: %U\n:END:")
                 :empty-lines 1)
                ("r" "reminder" entry
                 (file+headline ,agenda-path "Tasks & Appointments")
                 ,(concat "* %^{Description} :rem:\n%^t\n:PROPERTIES:\n"
                          ":CONTACT: %^{Who}\n:LOCATION: %^{Where}\n"
                          ":CREATED: %U\n:END:")
                 :empty-lines 1)
                ("b" "birthday" entry
                 (file+headline ,agenda-path "Birthdays & Recurrences")
                 ,(concat "* Compleanno %^{Who} :bday:\n"
                          "%(concat \"<\" (org-read-date) \" +1y>\")")
                 :empty-lines 1))))

      (setq org-blank-before-new-entry '((heading . t) (plain-list-item . nil)))
      (setq org-use-property-inheritance t)
      (setq org-hide-emphasis-markers t)
      (setq org-tags-column 0)
      (setq org-agenda-tags-column 0)

      ;; latex math display options
      (setq org-format-latex-options
            (plist-put org-format-latex-options :scale 1.8))
      (setq org-format-latex-options
            (plist-put org-format-latex-options :justify 'center))
      (setq org-preview-latex-default-process 'dvisvgm)
      (setq org-format-latex-options
            (plist-put org-format-latex-options :foreground "Black"))
      (setq org-format-latex-options
            (plist-put org-format-latex-options :background "Transparent"))

      (setq org-format-latex-header
            (concat org-format-latex-header "\n\\usepackage{melt-setup}"))

      (setq org-latex-default-packages-alist
            (assoc-delete-all "geometry" org-latex-default-packages-alist))

      (setq org-latex-packages-alist
            '(("margin=1in" "geometry" t)
              ("" "parskip" t)
              ("" "melt-setup" t)))

      (setq org-latex-images-centered t))
  #+end_src

*** Org Roam

  Provides non-linear personal wiki structures following the Zettelkasten note
  organization method.

  #+begin_src emacs-lisp
    ;; zettelkasten system
    (use-package org-roam
      :defer t
      :custom
      (org-roam-directory (expand-file-name org-dir-path))
      (org-roam-completion-everywhere t)
      :bind (("C-c n l" . org-roam-buffer-toggle)
             ("C-c n f" . org-roam-node-find)
             ("C-c n i" . org-roam-node-insert))
      :config
      (let* ((drawer (concat ":PROPERTIES:\n"
                             ":ID: %(org-id-new)\n"
                             ":CREATED: %U\n"
                             ":END:\n"))
             (base (concat drawer
                           "#+title: ${title}\n"
                           "#+author: " my/nickname "\n"))
             (book-header (concat ":PROPERTIES:\n"
                                  ":ID: %(org-id-new)\n"
                                  ":CREATED: %U\n"
                                  ":BOOK_TITLE: %^{Book Title}\n"
                                  ":END:\n"
                                  "#+title: ${title}\n"
                                  "#+author: " my/nickname "\n"
                                  "#+filetags: :uni:index:"))
             (person-template (concat drawer
                                      "#+title: ${title}\n"
                                      "#+filetags: :dream:person:\n\n"
                                      "* Dreams featuring ${title}\n")))

        (setq org-roam-capture-templates
              `(("u" "uni")
                ("uu" "uni" plain "%?"
                 :target (file+head "uni/${slug}.org"
                                    ,(concat base "#+filetags: :uni:"))
                 :unnarrowed t)
                ("ui" "index" plain "%?"
                 :target (file+head "uni/${slug}.org" ,book-header)
                 :unnarrowed t)

                ("d" "dream")
                ("dd" "dream" plain "%?"
                 :target (file+head "dreams/${slug}.org"
                                    ,(concat base "#+filetags: :dream:"))
                 :unnarrowed t)
                ("dp" "person" plain "%?"
                 :target (file+head "dreams/${slug}.org" ,person-template)
                 :unnarrowed t)

                ("s" "scrap" plain "%?"
                 :target (file+head "scraps/${slug}.org"
                                    ,(concat base "#+filetags: :scrap:"))
                 :unnarrowed t))))
      (org-roam-setup))
  #+end_src
