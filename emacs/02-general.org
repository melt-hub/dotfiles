** Indentation and Text Formatting

  Spacing, line wrapping rules, whitespace cleanup, and default column limits.

  #+begin_src emacs-lisp
    ;; Line Wrapping and Column Limits
    (global-visual-line-mode t)
    (setq-default fill-column 80)

    ;; Enable Hard Wrapping in Text and Programming Modes
    (add-hook 'text-mode-hook #'turn-on-auto-fill)
    (add-hook 'prog-mode-hook #'turn-on-auto-fill)

    ;; File Endings and Whitespace Cleanup
    (setq require-final-newline nil)
    (setq mode-require-final-newline nil)
    (add-hook 'before-save-hook #'delete-trailing-whitespace)

    ;; Indentation Rules (Spaces over Tabs)
    (setq-default tab-width 2)
    (setq-default indent-tabs-mode nil)

    ;; Language Offsets
    (setq-default lisp-indent-offset 2)
    (setq-default electric-indent-chars '(?\n ?\^? ?\}))

    ;; Buffer Text Scaling On File Open
    (add-hook 'find-file-hook (lambda () (text-scale-set 1)))

    (my/say "Indentation rules active. TAB: [%d], COL: [%d], LISP: [%d]."
            tab-width fill-column lisp-indent-offset)
  #+end_src

** File Management and Backups

  Centralize all temporary files (backups, auto-saves, and locks)
  to keep the file system clean while preventing data loss.

  #+begin_src emacs-lisp
    ;; Ensure Main Temp Directory Exists
    (unless (file-exists-p my/temp-dir)
      (make-directory my/temp-dir t))

    ;; Backups Configuration (file.org~)
    (setq backup-directory-alist
          `((".*" . ,(expand-file-name "backups/" my/temp-dir))))
    ; Preserve symlinks
    (setq backup-by-copying t)
    ; Use numbered backups
    (setq version-control t)
    ; Keep original version
    (setq kept-old-versions 2)
    ; Keep 5 most recent versions
    (setq kept-new-versions 5)
    ; Delete excess backups
    (setq delete-old-versions t)

    ;; Auto-Saves Configuration (#file.org#)
    (setq auto-save-file-name-transforms
          `((".*" ,(expand-file-name "auto-save/" my/temp-dir) t)))
    (setq auto-save-list-file-prefix
          (expand-file-name "auto-save-list/.saves-" my/temp-dir))

    ;; Lock Files Configuration (.#file.org)
    (when (boundp 'lock-file-name-transforms)
      (setq lock-file-name-transforms
            `((".*" ,(expand-file-name "locks/" my/temp-dir) t))))

    ;; Initialize Subdirectories
    (dolist (dir '("backups" "auto-save" "locks" "auto-save-list"))
      (let ((path (expand-file-name dir my/temp-dir)))
        (unless (file-exists-p path)
          (make-directory path t))))

    (my/say "Temporary files centralized in %s." my/temp-dir)
  #+end_src
