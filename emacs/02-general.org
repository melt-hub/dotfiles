#+TITLE: Emacs General Editor Configuration
#+PROPERTY: header-args :tangle "~/.emacs.d/modules/my-general.el"

* General

  This section configures global editor behavior, spacing parameters, status
  modelines, and standard formatting hooks.

  #+begin_src emacs-lisp
    ;; ====================| GENERAL |====================

    ;; modeline status indicators
    (line-number-mode t)
    (column-number-mode t)
    (display-time)

    ;; silence audible bell
    (setq ring-bell-function 'ignore)

    ;; suppress graphical dialogs, menus and modeline mouse interactions
    (setq use-dialog-box nil)
    (setq use-file-dialog nil)
    (setq context-menu-mode nil)
    (setq mode-line-major-mode-keymap nil)
    (setq mode-line-minor-mode-keymap nil)

    ;; text formatting and column limits
    (global-visual-line-mode t)
    (setq-default fill-column 80)
    (add-hook 'text-mode-hook #'turn-on-auto-fill)

    ;; posix file endings and whitespace cleanup
    (setq require-final-newline t)
    (add-hook 'before-save-hook #'delete-trailing-whitespace)

    ;; indentation defaults
    (setq-default tab-width 2)
    (setq-default indent-tabs-mode nil)
    (setq-default lisp-indent-offset 2)

    ;; default font size
    (set-face-attribute 'default nil :height 120)

    ;; centralized temporary files
    (unless (file-exists-p my/temp-dir)
      (make-directory my/temp-dir t))

    (setq backup-directory-alist `((".*" . ,(expand-file-name "backups/" my/temp-dir)))
          backup-by-copying t
          version-control t
          kept-old-versions 2
          kept-new-versions 5
          delete-old-versions t
          auto-save-file-name-transforms `((".*" ,(expand-file-name "auto-save/" my/temp-dir) t))
          auto-save-list-file-prefix (expand-file-name "auto-save-list/.saves-" my/temp-dir))

    (when (boundp 'lock-file-name-transforms)
      (setq lock-file-name-transforms `((".*" ,(expand-file-name "locks/" my/temp-dir) t))))

    ;; minibuffer prompts
    (setq use-short-answers t)
    (setq save-some-buffers-default-predicate 'save-some-buffers-root)

    ;; built in theme
    (use-package emacs
      :init
      (let ((inhibit-message t)
            (warning-minimum-level :emergency))
        (load-theme 'modus-vivendi t)))

    ;; module finalization
    (provide 'my-general)
    ;; ====================| END GENERAL |====================
  #+end_src
